53 SdlWindow(int32_t screenWidth, int32_t screenHeight, std::string title) {
54 /* SDL3 ne prend plus de position : le gestionnaire place la
55 * fenetre, et setPosition() reste la pour qui veut choisir. */
56 _window = SDL_CreateWindow(title.c_str(), screenWidth, screenHeight,
57 SDL_WINDOW_HIGH_PIXEL_DENSITY);
58 if (!_window)
59 return;
60
61 //nullptr : on laisse SDL choisir son pilote, Metal sur macOS
62 _renderer = SDL_CreateRenderer(_window, nullptr);
63 if (_renderer) {
64 //sinon un alpha < 255 est ecrase au lieu d'etre melange
65 SDL_SetRenderDrawBlendMode(_renderer, SDL_BLENDMODE_BLEND);
66 /* Une unite de dessin = un pixel logique, quel que soit
67 * l'ecran. Sans ca, sur un ecran retina la souris et le
68 * dessin ne parlent plus du meme repere - le meme piege que
69 * la vue chez sfml. */
70 SDL_SetRenderLogicalPresentation(_renderer, screenWidth, screenHeight,
71 SDL_LOGICAL_PRESENTATION_LETTERBOX);
72 }
73
74 _id = SDL_GetWindowID(_window);
75 registry()[_id] = this;
76 _lastTicks = SDL_GetTicks();
77 }

179
180 _delta = static_cast<int32_t>(now - _lastTicks);

183 * la cadence demandee, sinon la boucle tourne a vide et mange

185 if (_frameLimit > 0) {
186 const int32_t budget = 1000 / _frameLimit;
187
188 if (_delta < budget) {
189 SDL_Delay(static_cast<uint32_t>(budget - _delta));
190 _delta = budget;
191 }
192 }

222 static void pump() {
223 SDL_Event event;
224
225 /* Le curseur AVANT la file : l'etat suit le survol, et le survol
226 * a pu changer depuis la frame precedente meme sans evenement. */
227 applyCursor(SDL_GetMouseFocus());
228
229 while (SDL_PollEvent(&event)) {
230 const uint32_t target = windowOf(event);
231
232 if (target == 0) {
233 for (auto &[id, window] : registry())
234 window->push(event);
235 continue;
236 }
237
238 const auto found = registry().find(target);
239
240 if (found == registry().end())
241 continue; //une fenetre deja detruite : l'evenement tombe
242 found->second->push(event);
243 }
244 }

257 static void applyCursor(SDL_Window *focused) {
258 if (!focused)
259 return;
260
261 const auto found = registry().find(SDL_GetWindowID(focused));
262
263 if (found == registry().end())
264 return;
265
266 const bool wanted = found->second->_cursor;
267
268 if (wanted == shown())
269 return;
270 shown() = wanted;
271 if (wanted)
272 SDL_ShowCursor();
273 else
274 SDL_HideCursor();
275 }

285 static uint32_t windowOf(const SDL_Event &event) {
286 /* SDL3 a eclate SDL_WINDOWEVENT en un type par cas, tous ranges
287 * dans une plage : les prendre d'un bloc evite d'en oublier un
288 * a chaque version. */
289 if (event.type >= SDL_EVENT_WINDOW_FIRST && event.type <= SDL_EVENT_WINDOW_LAST)
290 return event.window.windowID;
291
292 switch (event.type) {
293 case SDL_EVENT_KEY_DOWN:
294 case SDL_EVENT_KEY_UP: return event.key.windowID;
295 case SDL_EVENT_MOUSE_BUTTON_DOWN:
296 case SDL_EVENT_MOUSE_BUTTON_UP: return event.button.windowID;
297 case SDL_EVENT_MOUSE_MOTION: return event.motion.windowID;
298 case SDL_EVENT_MOUSE_WHEEL: return event.wheel.windowID;
299 case SDL_EVENT_TEXT_INPUT: return event.text.windowID;
300 default: return 0;
301 }
302 }

313 void push(const SDL_Event &event) {
314 if (_events.size() >= MAX_EVENTS)
315 _events.erase(_events.begin());
316 _events.push_back(event);
317 feedEvent(event);
318 }
319
320 /* L'etat reconstruit DEPUIS LES EVENEMENTS, jamais par une requete
321 * globale comme SDL_GetKeyboardState() : celle-la est commune au
322 * processus, donc deux fenetres y liraient la meme chose meme si
323 * une seule a le focus.
324 *
325 * Seul ce qu'aucun evenement ne peut dire est garde : une touche
326 * reste "enfoncee" entre son KEYDOWN et son KEYUP. Les fronts, eux,
327 * se relisent dans _events, qui porte la frame. */
328 void feedEvent(const SDL_Event &event) {
329 switch (event.type) {
330 case SDL_EVENT_KEY_DOWN:
331 if (!event.key.repeat)
332 _keysDown[event.key.scancode] = true;
333 break;
334 case SDL_EVENT_KEY_UP:
335 _keysDown[event.key.scancode] = false;
336 break;
337 case SDL_EVENT_MOUSE_BUTTON_DOWN:
338 _mouseDown[event.button.button] = true;
339 break;
340 case SDL_EVENT_MOUSE_BUTTON_UP:
341 _mouseDown[event.button.button] = false;
342 break;
343 case SDL_EVENT_MOUSE_MOTION:
344 _mousePosition = {static_cast<double>(event.motion.x), static_cast<double>(event.motion.y)};
345 break;
346 case SDL_EVENT_WINDOW_FOCUS_LOST:
347 //sans ca, une touche relachee hors focus resterait "enfoncee"
348 _keysDown.fill(false);
349 _mouseDown.fill(false);
350 break;
351 default:
352 break;
353 }
354 }

378void SdlWindow::drawPoly(graphic::IPolygon *polygon) {
379 SdlPolygon *sdlPolygon = static_cast<SdlPolygon *>(polygon);
380
381 if (!_renderer || sdlPolygon->_vertices.empty())
382 return;
383
384 /* SDL_RenderGeometry n'a pas de transformation : il faut lui donner des
385 * coordonnees ecran. On recopie les sommets locaux en ajoutant la
386 * position, dans un tampon qui appartient au polygone - donc aucune
387 * allocation apres la premiere frame. */
388 const float x = static_cast<float>(sdlPolygon->_position.x);
389 const float y = static_cast<float>(sdlPolygon->_position.y);
390
391 for (size_t i = 0; i < sdlPolygon->_vertices.size(); i++) {
392 sdlPolygon->_screen[i] = sdlPolygon->_vertices[i];
393 sdlPolygon->_screen[i].position.x += x;
394 sdlPolygon->_screen[i].position.y += y;
395 }
396
397 SDL_RenderGeometry(_renderer, nullptr,
398 sdlPolygon->_screen.data(), static_cast<int>(sdlPolygon->_screen.size()),
399 nullptr, 0);
400}

438 *
439 * isKeyPressed / isKeyReleased : les evenements de la frame sont encore
440 * dans _events, donc un front s'y relit. Rien n'est consomme, donc
441 * n'importe quel nombre de lecteurs obtient la meme reponse.
442 * isKeyDown / isKeyUp : l'etat que la fenetre tient, puisqu'aucun
443 * evenement seul ne peut dire qu'une touche est toujours enfoncee.

447template <typename Match>
448static bool anyEvent(const std::vector<SDL_Event> &events, uint32_t type, Match match) {
449 for (const SDL_Event &event : events)
450 if (event.type == type && match(event))
451 return true;
452 return false;
453}
